                <h2 class="text-3xl font-bold tracking-tight mb-6">The Engineering Decisions</h2>

                <div class="segmented-control mb-8" id="moduleTabs">
                    <button class="segment-btn active" data-target="mod-api">Why +load?</button>
                    <button class="segment-btn" data-target="mod-discovery">Why namespace discovery?</button>

            <div class="reveal-hidden" data-reveal>

                <div id="mod-api" class="tab-pane">
                    <div class="grid md:grid-cols-2 gap-8 items-center">
                        <div>
                            <h3 class="text-xl font-semibold mb-4">Deferred Discovery via <code class="text-lg">+load</code></h3>
                            <p class="text-[var(--fg-muted)] text-sm mb-4">
                                The library activates itself automatically before <code>main()</code> runs. No developer integration code is required — merely linking the library runs <code>+load</code>.
                            </p>
                            <p class="text-[var(--fg-muted)] text-sm">
                                We defer the actual work to the next main-queue turn so UIKit is fully initialized, other libraries' <code>+load</code> methods have finished, and SwiftUI has registered its private classes with the ObjC runtime. The cost is a few milliseconds of launch latency; the benefit is that discovery runs in a settled runtime.
                            </p>
                        </div>
                        <div class="mac-window">
                            <div class="mac-titlebar">
                                <div class="traffic-lights"><div class="traffic-light tl-close"></div><div class="traffic-light tl-min"></div><div class="traffic-light tl-max"></div></div>
                                <div class="mac-title">QMTextLogger.m</div>
                            </div>
                            <div class="p-4 font-mono text-sm overflow-x-auto">
<pre><code><span class="syntax-keyword">@implementation</span> <span class="syntax-type">QMTextLogger</span>

<span class="syntax-macro">#pragma mark</span> + Load (zero-line integration)

+ (<span class="syntax-keyword">void</span>)load {
    <span class="syntax-comment">// Defer to ensure UIKit and SwiftUI</span>
    <span class="syntax-comment">// are fully registered with the runtime</span>
    <span class="syntax-function">dispatch_async</span>(dispatch_get_main_queue(), ^{
        [<span class="syntax-keyword">self</span> installHooks];
    });
}
<span class="syntax-keyword">@end</span></code></pre>
                            </div>

                <div id="mod-discovery" class="tab-pane hidden">
                    <div class="grid md:grid-cols-2 gap-8 items-center">
                        <div>
                            <h3 class="text-xl font-semibold mb-4">Mangled Name Discovery</h3>
                            <p class="text-[var(--fg-muted)] text-sm mb-4">
                                Hardcoding <code>_TtC7SwiftUI14CGDrawingView</code> breaks every iOS update. Instead, we enumerate all ObjC classes, filter for the SwiftUI namespace mangling prefix (<code>_TtC7SwiftUI</code> — that's <code>_TtC</code> for "class" plus length-prefixed module name <code>7SwiftUI</code>), and keep only <code>UIView</code> descendants.
                            </p>
                            <p class="text-[var(--fg-muted)] text-sm mb-4">
                                We then inspect their <em>ivars</em>. If they carry <code>NSString</code>, <code>NSAttributedString</code>, or a CoreText primitive, we flag them as text-bearing and swizzle their <code>drawRect:</code> — but only if the class <em>directly</em> implements it, never an inherited method.
                            </p>
                            <p class="text-[var(--fg-muted)] text-xs italic">
                                Swizzling an inherited <code>drawRect:</code> would mutate the superclass, affecting every <code>UIView</code> in the app. The <code>class_addMethod</code>-before-exchange pattern prevents that.
                            </p>
                        </div>
                        <div class="mac-window">
                            <div class="mac-titlebar">
                                <div class="traffic-lights"><div class="traffic-light tl-close"></div><div class="traffic-light tl-min"></div><div class="traffic-light tl-max"></div></div>
                                <div class="mac-title">QMTextLogger.m</div>
                            </div>
                            <div class="p-4 font-mono text-[13px] overflow-x-auto">
<pre><code><span class="syntax-keyword">static const char</span> kSwiftUIPrefix[] = <span class="syntax-string">"_TtC7SwiftUI"</span>;
<span class="syntax-keyword">static const size_t</span> kSwiftUIPrefixLen = <span class="syntax-function">sizeof</span>(kSwiftUIPrefix) - <span class="syntax-number">1</span>;

+ (<span class="syntax-keyword">void</span>)installHooks {
    <span class="syntax-keyword">unsigned int</span> count = <span class="syntax-number">0</span>;
    <span class="syntax-type">Class</span> *classes = <span class="syntax-function">objc_copyClassList</span>(&count);

    <span class="syntax-keyword">for</span> (<span class="syntax-keyword">int</span> i = <span class="syntax-number">0</span>; i < count; i++) {
        <span class="syntax-type">Class</span> cls = classes[i];
        <span class="syntax-keyword">const char</span> *name = <span class="syntax-function">class_getName</span>(cls);

        <span class="syntax-comment">// 1. Match SwiftUI namespace</span>
        <span class="syntax-keyword">if</span> (<span class="syntax-function">strncmp</span>(name, kSwiftUIPrefix, kSwiftUIPrefixLen) != <span class="syntax-number">0</span>) <span class="syntax-keyword">continue</span>;

        <span class="syntax-comment">// 2. Must be a view</span>
        <span class="syntax-keyword">if</span> (![cls isSubclassOfClass:[<span class="syntax-type">UIView</span> <span class="syntax-keyword">class</span>]]) <span class="syntax-keyword">continue</span>;

        <span class="syntax-comment">// 3. Ivar check &amp; direct-impl swizzle</span>
        <span class="syntax-keyword">if</span> ([<span class="syntax-keyword">self</span> classAppearsToRenderText:cls]) {
            [<span class="syntax-keyword">self</span> installDrawRectSwizzleOn:cls];
        }
    }
    <span class="syntax-function">free</span>(classes);
}</code></pre>
                            </div>
                        </div>
                    </div>
                </div>

                <div id="mod-dedup" class="tab-pane hidden">
                    <div class="grid md:grid-cols-2 gap-8 items-center">
                        <div>
                            <h3 class="text-xl font-semibold mb-4">Dedup via Associated Objects</h3>
                            <p class="text-[var(--fg-muted)] text-sm mb-4">
                                SwiftUI calls <code>drawRect:</code> on every render commit — not just when text changes. Without deduplication, we'd flood the notification stream on every animation tick and unrelated state update.
                            </p>
                            <p class="text-[var(--fg-muted)] text-sm">
                                We extract the text (favoring <code>accessibilityLabel</code> for speed, falling back to ivar walks when it's been overridden). Then we store the last-logged string on the view itself, keyed by <code>kQMLastTextKey</code>. Associated objects give us a "free ivar" on classes we don't own; storage is released automatically when the view dies.
                            </p>
                        </div>

    <section id="risks" class="py-20 px-6 bg-[var(--bg-page)] border-t border-[var(--border-light)]">
        <div class="max-w-5xl mx-auto reveal-hidden" data-reveal>
            <h2 class="text-3xl font-bold tracking-tight mb-8 text-center">Trade-offs & Mitigations</h2>

            <div class="grid md:grid-cols-2 gap-6">
                <div class="apple-card p-6">
                    <h3 class="font-semibold mb-2 flex items-center gap-2">
                        <svg class="w-5 h-5 text-[var(--apple-blue)]" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M10 20l4-16m4 4l4 4-4 4M6 16l-4-4 4-4"/></svg>
                        Heuristic vs. Surgical
                    </h3>
                    <p class="text-sm text-[var(--fg-muted)] mb-3">Why namespace prefixes and ivar heuristics instead of hardcoding <code>_TtC7SwiftUI14CGDrawingView</code>?</p>
                    <div class="text-xs font-medium text-[var(--apple-blue)] bg-blue-50 inline-block px-3 py-2 rounded leading-relaxed">
                        <strong>Mitigation:</strong> The hardcoded approach breaks every time Apple renames internal classes — which they've done every major iOS release since SwiftUI launched. The heuristic casts a wider net but survives rename churn. False positives are silent; false negatives are the cost of not hardcoding.
                    </div>
                </div>

                <div class="apple-card p-6">
                    <h3 class="font-semibold mb-2 flex items-center gap-2">
                        <svg class="w-5 h-5 text-[var(--apple-blue)]" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M13 16h-1v-4h-1m1-4h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z"/></svg>
                        App Store Safety
                    </h3>
                    <p class="text-sm text-[var(--fg-muted)] mb-3">Does introspecting private SwiftUI classes risk App Store rejection?</p>
                    <div class="text-xs font-medium text-[var(--apple-blue)] bg-blue-50 inline-block px-3 py-2 rounded leading-relaxed">
                        <strong>Mitigation:</strong> Calling private SPIs gets rejected. This library only uses <strong>public runtime APIs</strong> — <code>objc_copyClassList</code>, <code>class_addMethod</code>, <code>method_exchangeImplementations</code>. The same toolkit Firebase, Sentry, and every major analytics SDK uses. No private symbols are referenced by name.
                    </div>
                </div>

                <div class="apple-card p-6">
                    <h3 class="font-semibold mb-2 flex items-center gap-2">
                        <svg class="w-5 h-5 text-[var(--apple-blue)]" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M12 8v4l3 3m6-3a9 9 0 11-18 0 9 9 0 0118 0z"/></svg>
                        Performance Overhead
                    </h3>
                    <p class="text-sm text-[var(--fg-muted)] mb-3">The <code>drawRect:</code> hook runs on the CoreAnimation commit path, which SwiftUI hits frequently.</p>
                    <div class="text-xs font-medium text-[var(--apple-blue)] bg-blue-50 inline-block px-3 py-2 rounded leading-relaxed">
                        <strong>Mitigation:</strong> Per-view string dedup via an associated object. The common path — same text as last render — is one pointer dereference and one string compare before early return. No allocation, no notification post.
                    </div>
                </div>

                <div class="apple-card p-6">
                    <h3 class="font-semibold mb-2 flex items-center gap-2">
                        <svg class="w-5 h-5 text-[var(--apple-blue)]" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M9 12l2 2 4-4m5.618-4.016A11.955 11.955 0 0112 2.944a11.955 11.955 0 01-8.618 3.04A12.02 12.02 0 003 9c0 5.591 3.824 10.29 9 11.622 5.176-1.332 9-6.03 9-11.622 0-1.042-.133-2.052-.382-3.016z"/></svg>
                        Future Proofing
                    </h3>
                    <p class="text-sm text-[var(--fg-muted)] mb-3">What if SwiftUI moves off <code>drawRect:</code> to a pure <code>CALayer</code> rendering path in a future iOS release?</p>
                    <div class="text-xs font-medium text-[var(--apple-blue)] bg-blue-50 inline-block px-3 py-2 rounded leading-relaxed">
                        <strong>Mitigation:</strong> Known risk. If the substrate drops <code>drawRect:</code>, the library pivots to swizzling <code>-[CALayer display]</code> on the discovered layer classes — same namespace-discovery pattern, different hook point. The strategy is portable; only the hook selector changes.
                    </div>
                </div>
            </div>

                    captured: false,
                    ignoreReason: "UILabel lives outside the SwiftUI namespace. The prefix filter (_TtC7SwiftUI) excludes it by design.",
                    nodes: [
                        { name: 'UIWindow',         desc: 'The root window of the application.',                                                         addr: '0x107b3a100' },
                        { name: 'UIViewController', desc: 'Standard UIKit view controller managing the screen.',                                         addr: '0x107e33a10' },
                        { name: 'UIView',           desc: 'The root view of the view controller.',                                                       addr: '0x107e33b45' },
                        { name: 'UILabel',          desc: 'Standard UIKit label. Not in the SwiftUI namespace, so the namespace filter excludes it.',    addr: '0x107e8123c' }
                    ]
                }
            };

            let currentScenario = 'swiftui';

            const hierarchyExplorer = document.getElementById('hierarchyExplorer');
            const hierarchyDetail   = document.getElementById('hierarchyDetail');

            function renderHierarchy() {
                const scenario = scenarios[currentScenario];
                const data = scenario.nodes;
                if (!hierarchyExplorer) return;
                hierarchyExplorer.innerHTML = '';

                data.forEach((info, index) => {
                    const isTarget = scenario.captured && index === data.length - 1;
                    const node = document.createElement('div');
                    node.className = `flow-node ${isTarget ? 'active' : ''}`;
                    node.dataset.level = index;
                    node.innerHTML = `
                        <div class="font-mono text-[10px] ${isTarget ? 'text-[var(--apple-blue)]' : 'text-[var(--fg-muted)]'} uppercase tracking-wider mb-1">Level ${index}</div>
                        <div class="font-semibold text-sm ${isTarget ? 'text-[var(--apple-blue)]' : ''}">${info.name}</div>
                    `;
                    node.addEventListener('click', () => {
                        hierarchyExplorer.querySelectorAll('.flow-node').forEach(n => n.classList.remove('active'));
                        node.classList.add('active');
                        updateHierarchyDetails(index);
                    });
                    hierarchyExplorer.appendChild(node);

                    if (index < data.length - 1) {
                        const arrow = document.createElement('div');
                        arrow.innerHTML = `<svg class="w-5 h-5 text-[var(--border-medium)] hidden md:block" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M9 5l7 7-7 7"/></svg>`;
                        hierarchyExplorer.appendChild(arrow.firstChild);
                    }
                });

                updateHierarchyDetails(data.length - 1);
            }

            function updateHierarchyDetails(level) {
                const scenario = scenarios[currentScenario];
                const info = scenario.nodes[level];
                const isTarget = scenario.captured && level === scenario.nodes.length - 1;

                const currentTextHtml = isTarget
                    ? `<span class="text-[var(--apple-blue)] font-bold" id="hierarchyCurrentText">"${document.getElementById('sim-text').textContent}"</span>`
                    : `<span class="text-gray-400 italic">Not applicable — scenario is outside the library's scope</span>`;

                const header = isTarget
                    ? `${info.name} (Level ${level}) — Interception Target`
                    : `${info.name} (Level ${level})`;

                if (hierarchyDetail) {
                    hierarchyDetail.innerHTML = `
                        <h4 class="font-mono text-sm font-semibold mb-4 ${isTarget ? 'text-[var(--apple-blue)]' : 'text-gray-800'}">${header}</h4>
                        <p class="text-sm text-[var(--fg-muted)] mb-6">${info.desc}</p>
                        <div class="grid md:grid-cols-2 gap-4 text-sm font-mono text-gray-800">
                            <div><span class="text-[var(--fg-muted)] w-32 inline-block">Memory Addr:</span> ${info.addr}</div>
                            <div><span class="text-[var(--fg-muted)] w-32 inline-block">Current Text:</span> ${currentTextHtml}</div>
                        </div>
                    `;
                }
            }

            // Scenario tabs
            const scenarioTabsBtn = document.querySelectorAll('#scenarioTabs .segment-btn');
            const simBadge = document.getElementById('sim-badge');
            scenarioTabsBtn.forEach(tab => {
                tab.addEventListener('click', () => {
                    scenarioTabsBtn.forEach(t => t.classList.remove('active'));
                    tab.classList.add('active');
                    currentScenario = tab.dataset.scenario;

                    simBadge.textContent = tab.textContent;
                    renderHierarchy();

                    const simText = document.getElementById('sim-text');
                    simText.textContent = 'state: off';
                    simText.style.color = '#1d1d1f';

                    isToggled = false;
                    document.getElementById('sim-toggle').classList.remove('active');

                    const entry = document.createElement('div');
                    entry.className = 'mb-1 leading-snug break-all text-gray-300';
                    entry.innerHTML = `<span class="text-blue-400 italic">➜ Switched architecture to: ${tab.textContent}</span>`;
                    consoleOutput.appendChild(entry);
                    consoleOutput.scrollTop = consoleOutput.scrollHeight;
                });
            });

            renderHierarchy();

            // Simulator
            const toggle = document.getElementById('sim-toggle');
            const textDisplay = document.getElementById('sim-text');
            const consoleOutput = document.getElementById('console-output');
            const clearConsole = document.getElementById('clear-console');

            let isToggled = false;

            function appendLog(newVal) {
                const scenario = scenarios[currentScenario];
                const entry = document.createElement('div');
                entry.className = 'mb-1 leading-snug break-all';

                if (scenario.captured) {
                    entry.innerHTML = `<span class="text-[#34C759]">[QM]</span> text=<span class="text-white">${newVal}</span>  view=<span class="text-blue-300">&lt;${scenario.targetClass}:${scenario.ptrs.view}&gt;</span>  layer=<span class="text-purple-300">&lt;CALayer:${scenario.ptrs.layer}&gt;</span>  host=<span class="text-orange-300">&lt;${scenario.hostClass}:${scenario.ptrs.host}&gt;</span>`;
                    entry.classList.add('text-gray-300');
                } else {
                    entry.innerHTML = `<span class="text-gray-500 italic">— no capture — text changed on a UIKit element outside the namespace filter.</span><br><span class="text-gray-600 italic text-[10px]">${scenario.ignoreReason}</span>`;
                }

                consoleOutput.appendChild(entry);
                consoleOutput.scrollTop = consoleOutput.scrollHeight;
            }

            toggle.addEventListener('click', () => {
                isToggled = !isToggled;
                const newState = isToggled ? 'state: on' : 'state: off';

                toggle.classList.toggle('active', isToggled);
                textDisplay.textContent = newState;
                textDisplay.style.color = isToggled ? '#34C759' : '#1d1d1f';
                textDisplay.style.transform = 'scale(1.05)';
                setTimeout(() => textDisplay.style.transform = 'scale(1)', 150);

                const hierarchyCurrentText = document.getElementById('hierarchyCurrentText');
                if (hierarchyCurrentText) hierarchyCurrentText.textContent = `"${newState}"`;

                setTimeout(() => {
                    if (currentScenario === 'swiftui') {
                        const interceptRay = document.getElementById('intercept-ray');
                        if (interceptRay) {
                            interceptRay.style.opacity = '1';
                            setTimeout(() => interceptRay.style.opacity = '0', 500);
                        }
                    }
                    appendLog(newState);
                }, 50);
            });

            clearConsole.addEventListener('click', () => {
                consoleOutput.innerHTML = `
                    <div class="mb-4">
                        <div class="text-gray-400"><span class="text-[#34C759]">[QM]</span> Installed text-render hooks on 3 SwiftUI view class(es).</div>
                        <div class="text-gray-500 mt-1">Waiting for text renders...</div>
                    </div>
                `;
            });

            // Reveal animation + single-cycle architecture pulse
            let animationInterval;
            const observer = new IntersectionObserver((entries) => {
                entries.forEach((entry) => {
                    if (entry.isIntersecting) {
                        entry.target.classList.add('reveal-visible');

                        if (entry.target.querySelector('#pipelineDemo')) {
                            const nodes = ['node-swiftui', 'node-discovery', 'node-heuristic', 'node-hook'];
                            let activeIdx = 0;
                            if (animationInterval) clearInterval(animationInterval);
                            animationInterval = setInterval(() => {
                                nodes.forEach((id, idx) => {
                                    const el = document.getElementById(id);
                                    if (!el) return;
                                    if (idx === activeIdx) el.classList.add('active');
                                    else el.classList.remove('active');
                                });
                                activeIdx++;
                                if (activeIdx >= nodes.length) {
                                    clearInterval(animationInterval);
                                    setTimeout(() => {
                                        nodes.forEach(id => document.getElementById(id)?.classList.remove('active'));
                                        document.getElementById('node-hook')?.classList.add('active');
                                    }, 1000);
                                }
                            }, 800);
                        }

                        observer.unobserve(entry.target);
                    }
                });
            }, { threshold: 0.1, rootMargin: '0px 0px -50px 0px' });

            document.querySelectorAll('.reveal-hidden').forEach(el => observer.observe(el));
        });
    </script>
